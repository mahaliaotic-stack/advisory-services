Apps Script (Code.gs)

// ── Engagement Spike Model Intake ──────────────────────────────────────
// Spreadsheet: Advisory_Services_Accounts_Engagement_Spike_Models_Ionic_Migration_Tracker
// Tabs:  raw_sheet     (account_name | model_name | url | page_type)  ← submissions
//        all_accounts  (account_name in column A)                     ← dropdown source

const SHEET_NAME       = 'raw_sheet';
const ACCOUNTS_SHEET   = 'all_accounts';
const SPREADSHEET_NAME = 'Advisory_Services_Accounts_Engagement_Spike_Models_Ionic_Migration_Tracker';
// Recommended: paste the spreadsheet ID (from its URL) to skip the name lookup:
const SPREADSHEET_ID   = '';

// ---- WRITE: append submitted rows -------------------------------------
function doPost(e) {
  try {
    var data  = JSON.parse(e.postData.contents);
    var sheet = getSS_().getSheetByName(SHEET_NAME);
    if (!sheet) throw new Error('Tab not found: ' + SHEET_NAME);

    ensureHeaders_(sheet, ['account_name', 'model_name', 'url', 'page_type']);

    var rows = (data.rows || []).map(function (r) {
      return [ data.account_name || '', data.model_name || '', r.url || '', r.page_type || '' ];
    });
    if (rows.length) {
      sheet.getRange(sheet.getLastRow() + 1, 1, rows.length, 4).setValues(rows);
    }
    return json_({ status: 'success', inserted: rows.length });
  } catch (err) {
    return json_({ status: 'error', message: String(err) });
  }
}

// ---- READ: serve the account list (JSONP for the intake page) ---------
function doGet(e) {
  var cb     = e && e.parameter ? e.parameter.callback : null;
  var action = e && e.parameter ? e.parameter.action   : null;
  var payload = (action === 'accounts')
    ? { status: 'ok', accounts: getAccounts_() }
    : { status: 'ok', message: 'Engagement Spike intake endpoint is live' };

  var body = JSON.stringify(payload);
  if (cb) {
    return ContentService.createTextOutput(cb + '(' + body + ')')
      .setMimeType(ContentService.MimeType.JAVASCRIPT);
  }
  return ContentService.createTextOutput(body).setMimeType(ContentService.MimeType.JSON);
}

// ---- helpers ----------------------------------------------------------
function getSS_() {
  return SPREADSHEET_ID ? SpreadsheetApp.openById(SPREADSHEET_ID)
                        : openByName_(SPREADSHEET_NAME);
}

function getAccounts_() {
  var ss    = getSS_();
  var sheet = ss.getSheetByName(ACCOUNTS_SHEET);
  if (!sheet) return [];               // tab missing → page uses its fallback
  var last = sheet.getLastRow();
  if (last < 1) return [];
  var col = sheet.getRange(1, 1, last, 1).getValues()
    .map(function (r) { return String(r[0]).trim(); })
    .filter(function (v) { return v; });
  // Skip a header cell if the first value looks like a column label
  if (col.length && /^(account_name|account name|account|accounts|name)$/i.test(col[0])) {
    col.shift();
  }
  return col;
}

function openByName_(name) {
  var files = DriveApp.getFilesByName(name);
  if (files.hasNext()) return SpreadsheetApp.open(files.next());
  throw new Error('Spreadsheet not found: ' + name);
}

function ensureHeaders_(sheet, headers) {
  if (sheet.getLastRow() === 0) {
    sheet.getRange(1, 1, 1, headers.length).setValues([headers]);
  }
}

function json_(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}
