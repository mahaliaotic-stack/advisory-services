Apps Script (Code.gs)

// ── Engagement Spike Model Intake — backend ────────────────────────────
// Spreadsheet: Advisory_Services_Accounts_Engagement_Spike_Models_Ionic_Migration_Tracker
// Tabs:  raw_sheet     account_name | model_name | url | page_type | request   ← submissions
//        all_accounts  account_name (column A)                                 ← dropdown source
//        all_models    model_name (A) | … | domains (C)                        ← domain log (C2 down)
// Drive: ds_models/[Model].csv                        ← raw uploaded CSV
//        ionic_models/[Account]/[Model].csv           ← transformed CSV (url,page_type)

const SHEET_NAME       = 'raw_sheet';
const ACCOUNTS_SHEET   = 'all_accounts';
const MODELS_SHEET     = 'all_models';
const RAW_FOLDER       = 'ds_models';
const IONIC_FOLDER     = 'ionic_models';
const SPREADSHEET_NAME = 'Advisory_Services_Accounts_Engagement_Spike_Models_Ionic_Migration_Tracker';
const SPREADSHEET_ID   = '';   // recommended: paste the tracker's spreadsheet ID

// Source "Type" → transformed "page_type" (compared case-insensitively)
const TYPE_MAP = { 'highvalue': 'high_value', 'awareness': 'brand_awareness' };

// ─── WRITE: batch of models ────────────────────────────────────────────
function doPost(e) {
  try {
    var data    = JSON.parse(e.postData.contents);
    var account = String(data.account_name || '').trim();
    var request = String(data.request || '').trim();
    var models  = data.models;
    if (!models && data.rows) {              // legacy single-model shape
      models = [{ model_name: data.model_name, rows: data.rows }];
    }
    if (!Array.isArray(models) || !models.length) throw new Error('No models in payload');

    var ss    = getSS_();
    var sheet = ss.getSheetByName(SHEET_NAME);
    if (!sheet) throw new Error('Tab not found: ' + SHEET_NAME);
    ensureRawHeaders_(sheet);

    var rawFolder   = getOrCreateRootFolder_(RAW_FOLDER);
    var ionicParent = getOrCreateRootFolder_(IONIC_FOLDER);
    var acctFolder  = getOrCreateChildFolder_(ionicParent, account || 'Unspecified');

    var totalRows = 0, savedModels = 0;

    models.forEach(function (m) {
      var modelName = String(m.model_name || '').trim();
      if (!modelName) return;
      var rows = Array.isArray(m.rows) ? m.rows : [];

      // 1) append to raw_sheet  [account, model, url, raw type, request]
      if (rows.length) {
        var values = rows.map(function (r) {
          return [account, modelName, String(r.url || '').trim(), String(r.type || '').trim(), request];
        });
        sheet.getRange(sheet.getLastRow() + 1, 1, values.length, 5).setValues(values);
        totalRows += values.length;
      }

      // 2) unique domains → all_models column C, keyed by model_name
      var domains = Array.isArray(m.domains)
        ? m.domains
        : uniqueDomains_(rows.map(function (r) { return r.url; }));
      writeDomains_(ss, modelName, domains.join(', '));

      // 3) raw CSV → ds_models/[model].csv
      if (m.raw_csv != null) {
        saveCsv_(rawFolder, modelName + '.csv', String(m.raw_csv));
      }

      // 4) transformed CSV → ionic_models/[account]/[model].csv
      saveCsv_(acctFolder, modelName + '.csv', buildTransformed_(rows));

      savedModels++;
    });

    return json_({ status: 'success', models: savedModels, rows: totalRows });
  } catch (err) {
    return json_({ status: 'error', message: String(err) });
  }
}

// ─── READ: account list (JSONP) ────────────────────────────────────────
function doGet(e) {
  var cb     = e && e.parameter ? e.parameter.callback : null;
  var action = e && e.parameter ? e.parameter.action   : null;
  var payload = (action === 'accounts')
    ? { status: 'ok', accounts: getAccounts_() }
    : { status: 'ok', message: 'Engagement Spike intake endpoint is live' };

  var body = JSON.stringify(payload);
  if (cb) {
    return ContentService.createTextOutput(cb + '(' + body + ')')
      .setMimeType(ContentService.MimeType.JAVASCRIPT);
  }
  return ContentService.createTextOutput(body).setMimeType(ContentService.MimeType.JSON);
}

// ─── transforms ────────────────────────────────────────────────────────
function mapType_(t) {
  var key = String(t || '').trim();
  var mapped = TYPE_MAP[key.toLowerCase()];
  return (mapped != null) ? mapped : key.toLowerCase().replace(/\s+/g, '_');
}
function buildTransformed_(rows) {
  var out = 'url,page_type\n';
  rows.forEach(function (r) {
    out += csvCell_(r.url) + ',' + csvCell_(mapType_(r.type)) + '\n';
  });
  return out;
}
function csvCell_(v) {
  v = String(v == null ? '' : v);
  return /[",\n]/.test(v) ? '"' + v.replace(/"/g, '""') + '"' : v;
}
function uniqueDomains_(urls) {
  var seen = {}, out = [];
  (urls || []).forEach(function (u) {
    var d = domainOf_(u);
    if (d && !seen[d]) { seen[d] = 1; out.push(d); }
  });
  return out;
}
function domainOf_(u) {
  u = String(u || '').trim();
  if (!u || u.charAt(0) === '/') return null;
  u = u.replace(/^https?:\/\//i, '').replace(/^www\./i, '');
  var host = u.split(/[\/?#]/)[0];
  if (!host || host.indexOf('.') === -1) return null;
  return host.toLowerCase();
}

// ─── sheet + drive helpers ─────────────────────────────────────────────
function getSS_() {
  return SPREADSHEET_ID ? SpreadsheetApp.openById(SPREADSHEET_ID) : openByName_(SPREADSHEET_NAME);
}
function openByName_(name) {
  var f = DriveApp.getFilesByName(name);
  if (f.hasNext()) return SpreadsheetApp.open(f.next());
  throw new Error('Spreadsheet not found: ' + name);
}

function ensureRawHeaders_(sheet) {
  if (sheet.getLastRow() === 0) {
    sheet.getRange(1, 1, 1, 5).setValues([['account_name', 'model_name', 'url', 'page_type', 'request']]);
  } else if (!sheet.getRange(1, 5).getValue()) {
    sheet.getRange(1, 5).setValue('request');   // add new column header if it isn't there yet
  }
}

function getAccounts_() {
  var sheet = getSS_().getSheetByName(ACCOUNTS_SHEET);
  if (!sheet) return [];
  var last = sheet.getLastRow();
  if (last < 1) return [];
  var col = sheet.getRange(1, 1, last, 1).getValues()
    .map(function (r) { return String(r[0]).trim(); })
    .filter(function (v) { return v; });
  if (col.length && /^(account_name|account name|account|accounts|name)$/i.test(col[0])) col.shift();
  return col;
}

function writeDomains_(ss, modelName, domainsCsv) {
  var sheet = ss.getSheetByName(MODELS_SHEET);
  if (!sheet) {
    sheet = ss.insertSheet(MODELS_SHEET);
    sheet.getRange(1, 1).setValue('model_name');
    sheet.getRange(1, 3).setValue('domains');
  }
  var last = sheet.getLastRow(), rowIdx = -1;
  if (last >= 2) {
    var names = sheet.getRange(2, 1, last - 1, 1).getValues();
    for (var i = 0; i < names.length; i++) {
      if (String(names[i][0]).trim() === modelName) { rowIdx = i + 2; break; }
    }
  }
  if (rowIdx === -1) { rowIdx = Math.max(last + 1, 2); sheet.getRange(rowIdx, 1).setValue(modelName); }
  sheet.getRange(rowIdx, 3).setValue(domainsCsv);
}

function getOrCreateRootFolder_(name) {
  var it = DriveApp.getFoldersByName(name);
  return it.hasNext() ? it.next() : DriveApp.createFolder(name);
}
function getOrCreateChildFolder_(parent, name) {
  var it = parent.getFoldersByName(name);
  return it.hasNext() ? it.next() : parent.createFolder(name);
}
function saveCsv_(folder, filename, content) {
  var existing = folder.getFilesByName(filename);   // replace same-named file (idempotent)
  while (existing.hasNext()) existing.next().setTrashed(true);
  return folder.createFile(filename, content, MimeType.CSV);
}
function json_(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON);
}
  1. Open your project at script.google.com and replace Code.gs with the code above (delete the old contents first so there’s exactly one doPost/doGet).
  2. (Recommended) paste your tracker’s spreadsheet ID into SPREADSHEET_ID.
  3. New tabs/folders are created automatically on first use: all_models (domain log), Drive folders ds_models and ionic_models. A request column is added to raw_sheet.
  4. Redeploy: Deploy → Manage deployments → ✎ Edit → Version: New version → Deploy. The first time you’ll be asked to re-authorize (Drive access is new).
